\begin{frame}{Two-probe reduced dynamics}
Two spin-$1/2$ probes, $A$ and $B$, attach to opposite ends of a finite
XXZ bulk. The total Hamiltonian drives the dynamics; we follow only the probes.
\medskip
\begin{itemize}\setlength\itemsep{9pt}
\item \textbf{Ising contacts:} the probes act as conserved controls on the
bulk. The result is an exact, time-dependent dephasing (Schur) channel.
\item \textbf{Exchange contacts:} spin flows between probes and bulk.
Magnetization conservation splits the channel into five blocks.
\item \textbf{Texture:} a single probe input, $\ket{++}$, and a single
measurement setting.
\end{itemize}
\medskip
The slides give results and their meaning. All derivations are in the
accompanying report.
\end{frame}

\begin{frame}{Hamiltonian and normalization}
\small
$N$ bulk spins, $N+2$ total. Physical order: $A,1,\ldots,N,B$.
\[
S^\alpha=\frac{\sigma^\alpha}{2},\quad
S^\pm=S^x\pm iS^y,\quad \hbar=1.
\]
\[
H_E=J\sum_{j=1}^{N-1}
\bigl(S_j^xS_{j+1}^x+S_j^yS_{j+1}^y+\Delta S_j^zS_{j+1}^z\bigr).
\]
\begin{align*}
V_{\ZZ}&=g_A S_A^zS_1^z+g_B S_N^zS_B^z,\\
V_{\ex}&=\frac{g_A}{2}(S_A^+S_1^-+S_A^-S_1^+)
         +\frac{g_B}{2}(S_N^+S_B^-+S_N^-S_B^+),\\
H^{(r)}&=H_E+V_r,\qquad r\in\{\ZZ,\ex\}.
\end{align*}
$J,g_A,g_B$ have energy units. These are alternative contact models,
with no direct $A$--$B$ interaction and no applied fields in the reference model.
\end{frame}

\begin{frame}{A fixed product preparation defines a channel}
Group the Hilbert space as $\mathcal H_{AB}\otimes\mathcal H_E$.
Fix the same bulk density matrix $\omega_E$ for every probe input:
\[
\rho(0)=\rho_{AB}(0)\otimes\omega_E,\qquad
\boxed{\Lambda_t(X)=\Tr_E[U_t(X\otimes\omega_E)U_t^\dagger],\quad U_t=e^{-iH^{(r)}t}.}
\]
\begin{itemize}\setlength\itemsep{7pt}
\item $\Lambda_t$ is completely positive and trace preserving (CPTP) on the
four-dimensional two-probe space.
\item $\omega_E$ may be pure or mixed, thermal or not; changing it changes the channel.
\item Exact unitary dynamics plus a partial trace: no weak-coupling or Markov
approximation.
\item Initially correlated probe--bulk states have reduced trajectories,
but no universal channel.
\end{itemize}
\report{Sec.~2}
\end{frame}

% ==============================================================
\section{Ising contacts}
% ==============================================================

\begin{frame}{Ising coupling: the probes control the bulk}
Both probe spins are conserved, $[H^{(\ZZ)},S_A^z]=[H^{(\ZZ)},S_B^z]=0$.
For each configuration $a=(s_A,s_B)\in\{00,01,10,11\}$ the bulk feels
a different boundary field:
\[
H_a=H_E+g_As_AS_1^z+g_Bs_BS_N^z,\qquad
\boxed{U_t=\sum_a\ket a\bra a\otimes e^{-iH_at}.}
\]
\begin{itemize}\setlength\itemsep{7pt}
\item Four conditional bulk evolutions start from one common $\omega_E$.
\item The probes may occupy all four branches coherently.
\item Exact for any bulk Hamiltonian on $E$, even with transverse fields.
\end{itemize}
\report{Sec.~3.1}
\end{frame}

\begin{frame}{The reduced state: frozen populations, evolving coherences}
For an arbitrary probe input $\rho_{AB}(0)=(r_{ab})$,
\[
\boxed{[\rho_{AB}(t)]_{ab}=\Gamma_{ab}(t)\,r_{ab},\qquad
\Gamma_{ab}(t)=\Tr_E\bigl[\omega_E\,U_b(t)^\dagger U_a(t)\bigr].}
\]
\begin{itemize}\setlength\itemsep{7pt}
\item $\Gamma_{aa}=1$: the four populations never change.
\item $\Gamma_{ab}$ ($a\ne b$) is the overlap of two conditional bulk
histories. For a pure bulk, $|\Gamma_{ab}|^2$ is a conditional Loschmidt echo;
the channel also needs its phase.
\item Only six complex numbers evolve. They depend only on the boundary
difference $H_a-H_b=g_A\Delta s_A\,S_1^z+g_B\Delta s_B\,S_N^z$.
\item At $t=0$, $\dot\Gamma_{ab}$ is purely imaginary: the motion starts
coherently, and dephasing appears only later.
\end{itemize}
\report{Secs.~3.2 and 3.4}
\end{frame}

\begin{frame}{A legitimate dephasing channel}
\[
\boxed{\Lambda_t(X)=\Gamma(t)\circ X}\qquad(\circ:\ \text{entrywise product})
\]
\begin{itemize}\setlength\itemsep{6pt}
\item $\Gamma$ is a correlation matrix: $\Gamma\succeq0$, unit diagonal,
$|\Gamma_{ab}|\le1$. Hence CPTP and unital for every $\omega_E$.
\item Choi state $J_t=W\,(\Gamma/4)\,W^\dagger$, $W\ket a=\ket{aa}$: rank $\le4$.
\item Kraus operators are diagonal: $D_k=\sqrt{\lambda_k}\,\diag(v_k)$ from
$\Gamma=\sum_k\lambda_kv_kv_k^\dagger$.
\item Not depolarizing; not, in general, a product of single-probe channels.
\item Dephasing includes coherent phases and revivals. It need not be
irreversible.
\end{itemize}
\report{Secs.~3.3, 3.5 and 3.7}
\end{frame}

\begin{frame}{The $\ket{++}$ output is the whole channel}
Since $\ket{++}\bra{++}$ has all entries equal to $1/4$,
\[
\boxed{\rho_f(t)=\Lambda_t(\ket{++}\bra{++})=\tfrac14\Gamma(t),\qquad
\Lambda_t(X)=[4\rho_f(t)]\circ X.}
\]
\begin{itemize}\setlength\itemsep{7pt}
\item One input preparation, plus tomography of its six output coherences,
reconstructs the channel.
\item Populations are fixed at $1/4$, so eight local Pauli settings
$(\alpha,\beta)\ne(z,z)$ suffice.
\item Time derivatives of the same output give the time-local generator.
\item Special to the Ising family: it fails for general two-probe channels.
\end{itemize}
\report{Sec.~3.6}
\end{frame}

\begin{frame}{When can the channel be inverted?}
\[
\det\Lambda_t=\prod_{a<b}|\Gamma_{ab}(t)|^2 .
\]
\begin{itemize}\setlength\itemsep{6pt}
\item Invertible iff no coherence factor vanishes. Then the time-local
generator is entrywise:
\[
\dot\rho_{ab}=\eta_{ab}\rho_{ab},\qquad
\eta_{ab}=\frac{\dot\Gamma_{ab}}{\Gamma_{ab}}=-\kappa_{ab}+i\omega_{ab}.
\]
\item $\kappa_{ab}$ is the observed fractional damping rate of one
coherence; $\omega_{ab}$ is its phase rate.
\item Intermediate map $\Lambda_t\Lambda_s^{-1}$: Schur multiplication by
$\Gamma_{ab}(t)/\Gamma_{ab}(s)$. It is CPTP iff that matrix is $\succeq0$.
\item A coherence that vanishes and later revives rules out any
intermediate map through that time.
\end{itemize}
\report{Secs.~3.8 and 4.1}
\end{frame}

\begin{frame}{Canonical master equation: at most three rates}
At every invertible time, the canonical form of Hall \emph{et al.}~\cite{Hall2014} is
\[
\boxed{\dot\rho=-i[\Heff,\rho]+\sum_{k=1}^{3}\gamma_k(t)
\Bigl(L_k\rho L_k^\dagger-\tfrac12\{L_k^\dagger L_k,\rho\}\Bigr).}
\]
\begin{itemize}\setlength\itemsep{6pt}
\item $\Heff$ and all $L_k$ are diagonal combinations of
$\sigma_A^z,\ \sigma_B^z,\ \sigma_A^z\sigma_B^z$. No population transfer.
\item Twelve of the fifteen canonical rates vanish identically.
\item $\Heff$ is a bulk-induced configuration energy, which can include an
effective $\sigma_A^z\sigma_B^z$ coupling.
\item Everything follows in closed form from the six $\eta_{ab}(t)$ alone.
\end{itemize}
\report{Secs.~4.2--4.8}
\end{frame}

\begin{frame}{How the canonical data are obtained}
Let $F_\mu=\tfrac12(\sigma_A^z,\ \sigma_B^z,\ \sigma_A^z\sigma_B^z)$ and $G$ be the
$4\times3$ table of their eigenvalues $\pm\tfrac12$ ($G^{\mathsf T}G=I_3$,
$G^{\mathsf T}\mathbf1=0$). Then
\[
\boxed{\dmat=G^{\mathsf T}\eta\,G,\qquad
h_a=-\tfrac14\sum_b\operatorname{Im}\eta_{ab},\qquad
\gamma_k=\operatorname{spec}\dmat,\quad L_k=\textstyle\sum_\mu V_{\mu k}F_\mu.}
\]
\begin{itemize}\setlength\itemsep{6pt}
\item Only a $3\times3$ Hermitian matrix is diagonalized, never $16\times16$.
\item The solution is unique, so this \emph{is} the canonical form.
\item At $t=0$: $\dmat(0)=0$. The probes first see only the averaged
energies $\Tr(\omega_EH_a)$.
\end{itemize}
\report{Secs.~4.3--4.5, 4.8, 4.9 and 4.12}
\end{frame}

\begin{frame}{Physical reading: correlated damping and phase loops}
Write $\dmat=S+iA$ ($S$ real symmetric, $A$ real antisymmetric).
\begin{itemize}\setlength\itemsep{6pt}
\item \textbf{Damping:} $\kappa_{ab}=\tfrac12\Delta_{ab}^{\mathsf T}S\Delta_{ab}$
with $\Delta_{ab}=\mathbf g^a-\mathbf g^b$; the six $\kappa_{ab}$ fix $S$
uniquely.
\item Independent single-probe dephasing gives $S=\diag(2\kappa^A,2\kappa^B,0)$
and $A=0$. Any other entry signals dephasing \emph{correlated through the bulk}.
\item $S_{12}>0$: $00\leftrightarrow11$ decays faster than
$01\leftrightarrow10$ (common noise). $S_{12}<0$: anticorrelated noise.
\item \textbf{Phases:} $A$ is fixed by the three loop sums
$\omega_{ab}+\omega_{bc}+\omega_{ca}$. These are phases that no set of four
configuration energies can produce.
\end{itemize}
\report{Secs.~4.6 and 4.7}
\end{frame}

\begin{frame}{Negative rates, memory and singular times}
\begin{itemize}\setlength\itemsep{7pt}
\item Observed damping is bounded by the canonical rates:
$\gamma_{\min}\le\kappa_{ab}\le\gamma_{\max}$.
\item A coherence $|\Gamma_{ab}|$ that \emph{grows} implies a negative
canonical rate. The converse fails: $S_{33}<0$ also gives one.
\item CP divisibility $\iff\dmat(t)\succeq0$. A negative rate means
non-Markovian memory, \emph{not} an unphysical $\Lambda_t$, which stays CPTP.
\item At a zero of $\Gamma_{ab}$ the damping rate has a pole and some
$\gamma_k$ diverges, while the phase rate stays bounded.
\item $\rho_{AB}(t)$ itself remains smooth throughout.
\end{itemize}
\report{Secs.~4.10 and 4.11}
\end{frame}

\begin{frame}{Reference entanglement and probe entanglement}
\begin{itemize}\setlength\itemsep{7pt}
\item Unitary iff $\rank\Gamma=1$, i.e.\ $\Gamma_{ab}=u_au_b^*$: a diagonal
phase gate, possibly non-local.
\item Choi state versus reference:
$\mathcal N_{AB|R}=\tfrac14\sum_{a<b}|\Gamma_{ab}|$. The channel is
entanglement breaking iff every $\Gamma_{ab}=0$.
\item Between the probes:
\[
\boxed{\max_{\rho\in\mathrm{Sep}(A:B)}C[\Lambda_t(\rho)]=C[\rho_f(t)].}
\]
The channel can entangle $A$ with $B$ iff its $\ket{++}$ output is entangled,
and $\ket{++}$ is the optimal separable input.
\end{itemize}
\report{Secs.~3.7 and 3.9}
\end{frame}

\section{Exchange contacts}
% ==============================================================

\begin{frame}{Exchange coupling: spin flows, total magnetization survives}
\[
M_S=S_A^z+S_B^z,\quad M_E=\sum_{j=1}^NS_j^z,\qquad
[H^{(\ex)},M_S+M_E]=0.
\]
\begin{itemize}\setlength\itemsep{8pt}
\item Each contact moves one unit of $S^z$ between a probe and the bulk.
\item Probe populations and coherences can both change.
\item There is no four-branch controlled propagator.
\item A reduced-channel symmetry also needs a condition on the bulk
preparation.
\end{itemize}
\report{Sec.~5.1}
\end{frame}

\begin{frame}{Exchange channel: covariance and five blocks}
If the bulk preparation has no coherence between magnetization sectors,
$[\omega_E,M_E]=0$, then
\[
\boxed{\Lambda_t(e^{-i\theta M_S}Xe^{i\theta M_S})
=e^{-i\theta M_S}\Lambda_t(X)e^{i\theta M_S}.}
\]
Matrix units $\ket a\bra b$ have coherence order $q=m_a-m_b\in\{-2,\dots,2\}$,
with probe charges $(1,0,0,-1)$. Covariance forbids mixing different $q$:
\begin{center}
\small
\begin{tabular}{@{}ccl@{}}
\toprule
$q$&Size&Content\\\midrule
$0$&6&Populations and the $01\leftrightarrow10$ coherence\\
$\pm1$&4 each&Single-flip coherences\\
$\pm2$&1 each&$00\leftrightarrow11$ coherence\\\bottomrule
\end{tabular}
\end{center}
The Choi matrix has blocks of the same sizes (at most 70 real parameters,
not 256). Kraus operators move $0,\pm1,\pm2$ units of charge.
\report{Secs.~5.2--5.5}
\end{frame}

\begin{frame}{Fixed total charge gives an $X$-shaped state}
If the \emph{global} state lies in one total-$M$ sector, bulk kets of
different charge are orthogonal, and
\[
\rho_{AB}(t)=\begin{pmatrix}
p_{00}&0&0&0\\
0&p_{01}&c&0\\
0&c^*&p_{10}&0\\
0&0&0&p_{11}
\end{pmatrix},\qquad
C=2\max\{0,|c|-\sqrt{p_{00}p_{11}}\}.
\]
\begin{itemize}\setlength\itemsep{6pt}
\item Only the $01\leftrightarrow10$ coherence survives. It carries all
probe--probe entanglement.
\item With a fixed-charge bulk, $\ket{++}$ occupies \emph{three} total-charge
sectors, so its output need not have this form.
\end{itemize}
\report{Sec.~5.6}
\end{frame}

\section{Texture}
% ==============================================================

\begin{frame}{Quantum-state texture from one measurement setting}
Prepare $f_1=\ket{++}\bra{++}$, the textureless state of the
computational basis, and let $\rho_f(t)=\Lambda_t(f_1)$:
\[
p_f=\bra{++}\rho_f\ket{++}=\tfrac14\sum_{a,b}(\rho_f)_{ab},\qquad
\boxed{\Omega_{T_F}=1-p_f,\qquad \Omega_R=-\ln p_f}
\quad\text{\cite{Parisio2024,Vieira2026}.}
\]
\begin{itemize}\setlength\itemsep{6pt}
\item $p_f$ is the return probability. Measure both probes along $x$:
$p_f=P^{(x)}_{++}$.
\item Equivalently,
$4p_f=1+\langle\sigma_A^x\rangle+\langle\sigma_B^x\rangle+\langle\sigma_A^x\sigma_B^x\rangle$.
\item This holds in both regimes. Texture depends on phases, and is distinct
from purity and from entanglement.
\end{itemize}
\report{Sec.~2.4}
\end{frame}

\begin{frame}{Ising texture is a channel fidelity}
Using $\rho_f=\Gamma/4$:
\[
p_f=\frac14+\frac18\sum_{a<b}\operatorname{Re}\Gamma_{ab},\qquad
\boxed{F_e=p_f,\qquad F_{\rm avg}=\frac{4p_f+1}{5}}\quad\text{\cite{Nielsen2002}.}
\]
\begin{itemize}\setlength\itemsep{6pt}
\item These are fidelities of the two-probe map relative to the identity.
They are not $A\to B$ transfer fidelities.
\item The grand-sum response operator is $\mathcal R_t=\Gamma^{\mathsf T}$.
\item $\Omega_{T_F}=0$ only for the identity channel within the Ising family.
\item One real number: it does not fix the six coherences, purity,
entanglement or divisibility.
\end{itemize}
\report{Secs.~3.10 and 3.11}
\end{frame}

\begin{frame}{Exchange texture: what the same input determines}
With a magnetization-invariant bulk, $f_1$ probes all five coherence orders.
\begin{itemize}\setlength\itemsep{7pt}
\item One output fixes each block's action on \emph{one vector} only. The
$6$- and $4$-dimensional blocks remain undetermined.
\item Exception, the one-dimensional sector:
$\Lambda_t(\ket{00}\bra{11})=\xi_{+2}(t)\ket{00}\bra{11}$, with
$\boxed{\xi_{+2}=4[\rho_f]_{00,11}}$.
\item $F_e=p_f$ requires the Schur structure; $F_{\rm avg}=(4F_e+1)/5$
always holds.
\end{itemize}
\report{Sec.~5.7}
\end{frame}

\begin{frame}{Summary}
\begin{center}
\small
\begin{tabular}{@{}p{2.7cm}p{4.9cm}p{5.0cm}@{}}
\toprule
&Ising contacts&Exchange contacts\\\midrule
Map structure&$\Gamma\circ X$, exact&Five coherence-order blocks\\[4pt]
Populations&Frozen&Can move\\[4pt]
Generator&$\le3$ rates, diagonal $L_k$, closed form&Block entries from dynamics\\[4pt]
Information in $\rho_f$&Complete channel&Partial block information\\[4pt]
Assumptions&Any fixed $\omega_E$, even with transverse fields&$[H,M]=0$ and $[\omega_E,M_E]=0$\\\bottomrule
\end{tabular}
\end{center}
\medskip
Symmetry fixes which probe operators can mix. The time dependence of the
coefficients requires the many-body bulk dynamics.
\end{frame}

\begin{frame}{Conventions}
\small
\begin{itemize}\setlength\itemsep{6pt}
\item Probe entropy $S(\rho)=-\Tr\rho\ln\rho$ in nats. For a globally pure
state, $S(\rho_{AB})$ measures entanglement across $AB|E$.
\item Concurrence and negativity concern $A|B$. For two qubits,
$C>0\iff\mathcal N>0$ \cite{Wootters1998}.
\item The normalized Choi state has trace one. Its reference--output
entanglement is distinct from entanglement between the probes.
\item Notation: $\Gamma_{ab}$ coherence factors, $\eta_{ab}$ their
logarithmic derivatives, $\gamma_k$ canonical rates, $\xi_{\pm2}$ exchange
multipliers.
\end{itemize}
\end{frame}
